package com.kausar.audiokuran;

import android.app.Notification;
import android.app.NotificationChannel;
import android.app.NotificationManager;
import android.app.PendingIntent;
import android.app.Service;
import android.content.Context;
import android.content.Intent;
import android.media.MediaPlayer;
import android.os.Build;
import android.os.Handler;
import android.os.IBinder;
import android.os.Looper;

public class AudioService extends Service {
    private static AudioService instance;
    private NotificationManager nm;
    private static final String CHANNEL_ID = "quran_playback";
    private static final int NOTIF_ID = 1001;
    private final Handler handler = new Handler(Looper.getMainLooper());
    private final Runnable ticker = new Runnable(){
        public void run(){
            try { updateNotification(); } catch(Exception e){}
            handler.postDelayed(this, 2000);
        }
    };

    public static AudioService getInstance(){ return instance; }

    @Override public IBinder onBind(Intent i){ return null; }

    @Override
    public void onCreate(){
        super.onCreate();
        instance = this;
        nm = (NotificationManager) getSystemService(Context.NOTIFICATION_SERVICE);
        if(Build.VERSION.SDK_INT >= Build.VERSION_CODES.O){
            NotificationChannel ch = new NotificationChannel(CHANNEL_ID, "Pemutaran Al-Quran", NotificationManager.IMPORTANCE_LOW);
            ch.setDescription("Kontrol pemutaran audio Al-Quran");
            ch.setShowBadge(false);
            nm.createNotificationChannel(ch);
        }
        startForeground(NOTIF_ID, buildNotification());
        handler.postDelayed(ticker, 2000);
    }

    @Override
    public int onStartCommand(Intent intent, int flags, int startId){
        if(intent != null && intent.getAction() != null){
            String a = intent.getAction();
            PlayerManager pm = MainActivity.getStaticPm();
            if(pm != null){
                MediaPlayer pl = pm.getPlayer();
                try {
                    if("PLAY".equals(a)){
                        if(!pl.isPlaying()){ if(pl.getDuration()>0) pl.start(); else pm.play(pm.state.surah, pm.state.ayat); }
                    } else if("PAUSE".equals(a)){
                        if(pl.isPlaying()) pl.pause();
                    } else if("NEXT".equals(a)){
                        pm.play(pm.state.surah, pm.state.ayat + 1);
                    } else if("PREV".equals(a)){
                        pm.play(pm.state.surah, pm.state.ayat - 1);
                    } else if("STOP".equals(a)){
                        if(pl.isPlaying()) pl.stop();
                        pl.reset();
                        pm.cancelSleepTimer();
                        handler.removeCallbacksAndMessages(null);
                        stopForeground(true);
                        stopSelf();
                        return START_NOT_STICKY;
                    }
                } catch(Exception e){}
            }
        }
        updateNotification();
        return START_NOT_STICKY;
    }

    @Override
    public void onDestroy(){
        handler.removeCallbacksAndMessages(null);
        instance = null;
        super.onDestroy();
    }

    private PendingIntent actionIntent(String action){
        Intent i = new Intent(this, AudioService.class);
        i.setAction(action);
        int flag = PendingIntent.FLAG_UPDATE_CURRENT;
        if(Build.VERSION.SDK_INT >= Build.VERSION_CODES.M) flag |= PendingIntent.FLAG_IMMUTABLE;
        return PendingIntent.getService(this, action.hashCode(), i, flag);
    }

    public void updateNotification(){
        try { nm.notify(NOTIF_ID, buildNotification()); } catch(Exception e){}
    }

    private Notification buildNotification(){
        Intent open = new Intent(this, MainActivity.class);
        open.setFlags(Intent.FLAG_ACTIVITY_SINGLE_TOP | Intent.FLAG_ACTIVITY_CLEAR_TOP);
        int flag = PendingIntent.FLAG_UPDATE_CURRENT;
        if(Build.VERSION.SDK_INT >= Build.VERSION_CODES.M) flag |= PendingIntent.FLAG_IMMUTABLE;
        PendingIntent pi = PendingIntent.getActivity(this, 0, open, flag);

        PlayerManager pm = MainActivity.getStaticPm();
        String title = "Alquran Indonesia V2";
        String text = "Siap memutar";
        boolean isPlaying = false;
        if(pm != null){
            try {
                Data.Surah s = Data.SURAH[pm.state.surah-1];
                title = s.name + " - Ayat " + pm.state.ayat;
                text = pm.getReciterName(pm.state.reciter);
                isPlaying = pm.getPlayer().isPlaying();
            } catch(Exception e){}
        }

        Notification.Builder b;
        if(Build.VERSION.SDK_INT >= Build.VERSION_CODES.O){
            b = new Notification.Builder(this, CHANNEL_ID);
        } else {
            b = new Notification.Builder(this);
        }
        b.setContentTitle(title)
         .setContentText(text)
         .setSmallIcon(android.R.drawable.ic_media_play)
         .setContentIntent(pi)
         .setOngoing(isPlaying)
         .setOnlyAlertOnce(true)
         .setShowWhen(false);

        b.addAction(new Notification.Action.Builder(
            android.R.drawable.ic_media_previous, "Sebelumnya", actionIntent("PREV")).build());

        if(isPlaying){
            b.addAction(new Notification.Action.Builder(
                android.R.drawable.ic_media_pause, "Jeda", actionIntent("PAUSE")).build());
        } else {
            b.addAction(new Notification.Action.Builder(
                android.R.drawable.ic_media_play, "Putar", actionIntent("PLAY")).build());
        }

        b.addAction(new Notification.Action.Builder(
            android.R.drawable.ic_media_next, "Selanjutnya", actionIntent("NEXT")).build());

        b.addAction(new Notification.Action.Builder(
            android.R.drawable.ic_menu_close_clear_cancel, "Tutup", actionIntent("STOP")).build());

        b.setStyle(new Notification.MediaStyle().setShowActionsInCompactView(0, 1, 2));

        return b.build();
    }
}

            staticPm.state.surah = storage.getSurah();

            String r = storage.getReciter(); boolean valid=false;
            for(Data.Reciter rr: Data.RECITERS) if(rr.code.equals(r)){ valid=true; break; }

            staticPm.state.repeatMode = storage.getRepeatMode();
            staticPm.state.volume = storage.getVolume();
        }
        pm = staticPm;
        pm.setListener(() -> runOnUi(() -> updateAllViews()));

        container = new FrameLayout(this);
        container.setBackgroundColor(getResources().getColor(R.color.colorBackground));
        setContentView(container);

        if(Build.VERSION.SDK_INT >= 33){
            try {
                if(checkSelfPermission("android.permission.POST_NOTIFICATIONS") != android.content.pm.PackageManager.PERMISSION_GRANTED){
                    requestPermissions(new String[]{"android.permission.POST_NOTIFICATIONS"}, 991);
                }
            } catch(Exception e){}
        }
        try { startForegroundService(new Intent(this, AudioService.class)); }
        catch(Exception e){ try { startService(new Intent(this, AudioService.class)); } catch(Exception e2){} }

        Utils.toast(this, getString(R.string.toast_developer), true);
        TerjemahanCache.fetch(storage.getSurah(), null);
        if(savedInstanceState != null){
            currentMenu = savedInstanceState.getInt("currentMenu", 0);
            int[] stackInt = savedInstanceState.getIntArray("menuStack");
            if(stackInt != null) for(int x: stackInt) menuStack.push(x);
            showMenu(currentMenu);
        } else {
            showMenu(0);
        }
    }

    @Override
    protected void onSaveInstanceState(Bundle b){
        super.onSaveInstanceState(b);
        b.putInt("currentMenu", currentMenu);
        Integer[] arr = menuStack.toArray(new Integer[0]);
        int[] out = new int[arr.length];
        for(int i=0;i<arr.length;i++) out[i] = arr[i];
        b.putIntArray("menuStack", out);
    }

    private void goBack(){
        if(currentDialog != null){ try { currentDialog.dismiss(); } catch(Exception e){} currentDialog = null; return; }
        if(menuStack.isEmpty()) performExit();
        else showMenu(menuStack.pop());
    }

    private void performExit(){
        Utils.speak(getString(R.string.toast_thanks));
        Utils.toast(this, getString(R.string.toast_thanks), false);
        finish();

    private void showMenu(int id){
        currentMenu = id;
        if(currentDialog != null){
            try { currentDialog.dismiss(); } catch(Exception e){}
            currentDialog = null;
        }
        View v;
        switch(id){
            case 1: v = buildSurahList(); break;
            case 2: v = buildJuzList(); break;
            case 3: v = buildSleepRange(); break;
            case 4: v = buildPlayerDialog(); break;
            case 5: v = buildAudioSettings(); break;
            case 6: v = buildReciterDialog(); break;
            case 7: v = buildAbout(); break;
            case 8: v = buildResetConfirm(); break;
            case 9: v = buildDoaList(); break;
            case 10: v = buildDoaDetail(currentDoa); break;
            default:
                container.removeAllViews();
                v = buildMainMenu();
                container.addView(v);
                updateAllViews();
                return;
        }
        Dialog d = new Dialog(this);
        d.requestWindowFeature(Window.FEATURE_NO_TITLE);
        d.setContentView(v);
        Window w = d.getWindow();
        if(w != null){
            w.setLayout(android.view.WindowManager.LayoutParams.MATCH_PARENT, android.view.WindowManager.LayoutParams.MATCH_PARENT);
            w.setBackgroundDrawableResource(android.R.color.white);
            w.setSoftInputMode(android.view.WindowManager.LayoutParams.SOFT_INPUT_ADJUST_RESIZE);
        }
        d.setOnCancelListener(dd -> { currentDialog = null; });
        currentDialog = d;
        d.show();
        updateAllViews();
    }

    private void updateAllViews(){
        if(txtStatus==null) return;
        Data.Surah s = Data.SURAH[pm.state.surah-1];
        String qariName = pm.getReciterName(pm.state.reciter);
        StringBuilder sb = new StringBuilder();
        sb.append(String.format(Locale.US,"Qari: %s\nPosisi: %s (Ayat %d dari %d)", qariName, s.name, pm.state.ayat, s.verses));
        if(pm.state.rangePlayActive)
            sb.append(String.format(Locale.US,"\n[Mode Tidur: Surah %s (%d) - Surah %s (%d) AKTIF]",
                Data.SURAH[pm.state.startSurah-1].name,pm.state.startAyat,
                Data.SURAH[pm.state.endSurah-1].name,pm.state.endAyat));
        if(pm.state.sleepTimerActive) sb.append("\n[Sleep Timer: AKTIF]");
        String repeatLabel = getString(R.string.repeat_off);
        if("ayat".equals(pm.state.repeatMode)) repeatLabel = getString(R.string.repeat_ayat);
        else if("surah".equals(pm.state.repeatMode)) repeatLabel = getString(R.string.repeat_surah);
        else if("juz".equals(pm.state.repeatMode)) repeatLabel = getString(R.string.repeat_juz);
        sb.append("\n[Mode Ulangi: ").append(repeatLabel).append("]");
        txtStatus.setText(sb.toString());
        if(pm.state.surah>0 && pm.state.surah<=114 && !TerjemahanCache.isLoaded(pm.state.surah)){ TerjemahanCache.fetch(pm.state.surah, null); }
        if(btnPlayPause!=null){
            boolean playing = false;
            try { playing = player.isPlaying(); } catch(Exception e){}
            btnPlayPause.setText(playing ? R.string.btn_pause : R.string.btn_play);
        }
    }

    private void updatePlayerStatus(TextView tv,Button bP){
        Data.Surah s=Data.SURAH[pm.state.surah-1];
        String qariName=pm.getReciterName(pm.state.reciter);
        StringBuilder sb=new StringBuilder();
        sb.append(String.format(Locale.US,"Qari: %s\n%s\nAyat %d dari %d (Juz %d)",qariName,s.name,pm.state.ayat,s.verses,s.juz));
        if(pm.state.rangePlayActive)
            sb.append(String.format(Locale.US,"\n[Mode Tidur: Surah %s (%d) - Surah %s (%d) AKTIF]",
                Data.SURAH[pm.state.startSurah-1].name,pm.state.startAyat,
                Data.SURAH[pm.state.endSurah-1].name,pm.state.endAyat));
        if(pm.state.sleepTimerActive) sb.append("\n[Sleep Timer: AKTIF]");
        tv.setText(sb.toString());
        boolean playing = false;
        try { playing = player.isPlaying(); } catch(Exception e){}
        bP.setText(playing ? R.string.btn_pause : R.string.btn_play);
    }

##### JAVA: PlayerManager.java #####
package com.kausar.audiokuran;

import android.content.Context;
import android.media.AudioManager;
import android.media.MediaPlayer;
import android.os.Handler;
import android.os.Looper;
import java.util.Locale;

public class PlayerManager {
    public interface Listener { void onUpdate(); }
    public static String[] AUDIO_SOURCES = new String[]{
        "https://everyayah.com/data/%s/%03d%03d.mp3",
        "https://download.quranicaudio.com/quran/%s/%03d%03d.mp3",
        "https://server8.mp3quran.net/%s/%03d%03d.mp3"
    };
    public static class State {
        public int surah, ayat;
        public String reciter="Alafasy_128kbps";
        public String repeatMode="off";
        public int volume=100;
        public boolean rangePlayActive=false;
        public int startSurah=2,startAyat=1,endSurah=2,endAyat=142;
        public boolean isExiting=false, sleepTimerActive=false;
        public int retryCount=0;
    }
    public State state = new State();
    private MediaPlayer player; private Context ctx; private Storage storage;
    private Listener listener; private Handler handler = new Handler(Looper.getMainLooper());
    private Runnable sleepRunnable;

    public PlayerManager(Context ctx, Storage st, MediaPlayer p, Listener l){
        this.ctx=ctx; this.storage=st; this.player=p; this.listener=l;
    }
    public MediaPlayer getPlayer(){ return player; }
    public void setListener(Listener l){ this.listener=l; }

    public String getReciterName(String code){
        for(Data.Reciter r: Data.RECITERS) if(r.code.equals(code)) return r.name;
        return "Mishary Rashid Alafasy";
    }
    public int getJuzIndex(int surah,int ayat){
        for(int i=29;i>=0;i--){ Data.JuzStart js=Data.JUZ_START[i];
            if(surah>js.surah||(surah==js.surah&&ayat>=js.ayat)) return i+1; }
        return 1;
    }
    public int[] getNextPosition(int surah,int ayat){
        int max=Data.SURAH[surah-1].verses;
        if(ayat<max) return new int[]{surah,ayat+1};
        else if(surah<114) return new int[]{surah+1,1};
        else return new int[]{surah,max+1};
    }
    public int comparePosition(int s1,int a1,int s2,int a2){
        if(s1!=s2) return s1<s2?-1:1;
        if(a1!=a2) return a1<a2?-1:1;
        return 0;
    }
    public String getUrl(int surah,int ayat,int sourceIndex){
        if("Muhammad_Ayyoub_Full".equals(state.reciter))
            return String.format(Locale.US,"https://server8.mp3quran.net/ayyub/%03d.mp3",surah);
        if(sourceIndex<1||sourceIndex>AUDIO_SOURCES.length) return null;
        return String.format(Locale.US,AUDIO_SOURCES[sourceIndex-1],state.reciter,surah,ayat);
    }
    public void applyVolume(){
        try { float v=state.volume/100.0f; player.setVolume(v,v); } catch(Exception e){}
    }
    public void startSleepTimer(int minutes){
        cancelSleepTimer();
        if(minutes>0){
            state.sleepTimerActive=true;
            sleepRunnable = () -> {
                state.sleepTimerActive=false; state.rangePlayActive=false;
                try { if(player.isPlaying()) player.stop(); player.reset(); } catch(Exception e){}
                Utils.speak("Waktu tidur selesai. Pemutaran dihentikan.");
                Utils.toast(ctx,"Sleep timer selesai.",false);
                if(listener!=null) listener.onUpdate();
            };
            handler.postDelayed(sleepRunnable, minutes*60L*1000L);
        }
    }
    public void cancelSleepTimer(){
        if(sleepRunnable!=null){ handler.removeCallbacks(sleepRunnable); sleepRunnable=null; }
        state.sleepTimerActive=false;
    }
    public void play(int surahIn,int ayatIn){ play(surahIn,ayatIn,1,false); }
    public void play(int surahIn,int ayatIn,int sourceIndex,boolean isRetry){
        int surah=surahIn, ayat=ayatIn;
        if(surah<1) surah=1;
        if(surah>Data.SURAH.length) surah=Data.SURAH.length;
        int maxAyat = Data.SURAH[surah-1].verses;
        if("Muhammad_Ayyoub_Full".equals(state.reciter)) ayat=1;
        else {
            if(ayat<1){ if(surah>1){surah=surah-1;ayat=Data.SURAH[surah-1].verses;} else ayat=1; }
            else if(ayat>maxAyat){
                if(surah<Data.SURAH.length){surah=surah+1;ayat=1;}
                else {
                    if(state.rangePlayActive){
                        state.rangePlayActive=false;
                        Utils.speak("Pemutaran rentang ayat selesai.");
                        Utils.toast(ctx,"Pemutaran selesai.",false);
                        if(listener!=null) listener.onUpdate(); return;
                    }
                    Utils.speak("Pembacaan Al-Qur'an selesai."); return;
                }
            }
        }
        if(state.rangePlayActive){
            if(comparePosition(surah,ayat,state.endSurah,state.endAyat)>0){
                state.rangePlayActive=false;
                try { if(player.isPlaying()) player.stop(); player.reset(); } catch(Exception e){}
                Utils.speak("Pemutaran sampai Surah "+Data.SURAH[state.endSurah-1].name+" ayat "+state.endAyat+" telah selesai. Selamat tidur.");
                Utils.toast(ctx,String.format(Locale.US,"Pemutaran otomatis berhenti (Mencapai batas Surah %s Ayat %d).",Data.SURAH[state.endSurah-1].name,state.endAyat),true);
                if(listener!=null) listener.onUpdate(); return;
            }
        }
        state.surah=surah; state.ayat=ayat;
        storage.saveState(surah,ayat);
        if(listener!=null) listener.onUpdate();
        if(!isRetry) state.retryCount=0;

        final String url = getUrl(surah,ayat,sourceIndex);
        if(url==null){
            try { player.reset(); } catch(Exception e){}
            Utils.toast(ctx,"Semua sumber audio gagal. Periksa koneksi atau pilih qari lain.",true);
            Utils.speak("Audio tidak tersedia untuk qari ini. Silakan pilih qari lain.");
            state.rangePlayActive=false; cancelSleepTimer();
            if(listener!=null) listener.onUpdate(); return;
        }
        try {
            player.reset();
            player.setAudioStreamType(AudioManager.STREAM_MUSIC);
            player.setDataSource(url);
            player.setOnPreparedListener(mp -> {
                state.retryCount=0;
                float v=state.volume/100.0f; mp.setVolume(v,v); mp.start();
                if(listener!=null) listener.onUpdate();
            });
            player.setOnCompletionListener(mp -> onComplete());
            final int fSurah = surah; final int fAyat = ayat; final int fSource = sourceIndex;
            player.setOnErrorListener((mp,w,e) -> {
                state.retryCount++;
                if(state.retryCount<AUDIO_SOURCES.length){
                    int nextSource=fSource+1;
                    Utils.toast(ctx,"Gagal memutar dari sumber ke-"+sourceIndex+". Mencoba sumber ke-"+nextSource+"...",true);
                    Utils.speak("Mencoba sumber audio alternatif.");
                    play(fSurah,fAyat,nextSource,true);
                } else {
                    Utils.toast(ctx,"Semua sumber audio gagal untuk ayat ini. Periksa koneksi atau pilih qari lain.",true);
                    Utils.speak("Audio tidak tersedia untuk qari ini. Silakan pilih qari lain.");
                    try { mp.reset(); } catch(Exception ex){}
                    state.rangePlayActive=false; cancelSleepTimer();
                    if(listener!=null) listener.onUpdate();
                }
                return true;
            });
            player.prepareAsync();
        } catch(Exception e){ Utils.toast(ctx,"Error: "+e.getMessage(),true); }
    }
    private void onComplete(){
        if(state.sleepTimerActive && !state.rangePlayActive){
            cancelSleepTimer();
            try { player.reset(); } catch(Exception e){}
            Utils.speak("Waktu tidur selesai. Pemutaran dihentikan.");
            Utils.toast(ctx,"Sleep timer selesai.",false);
            if(listener!=null) listener.onUpdate(); return;
        }
        String mode = state.repeatMode; int ns,na;
        if("Muhammad_Ayyoub_Full".equals(state.reciter)){
            if("surah".equals(mode)||"ayat".equals(mode)){ ns=state.surah; na=1; }
            else {
                ns=state.surah+1;
                if(ns>114){
                    if(state.rangePlayActive){
                        state.rangePlayActive=false;
                        Utils.speak("Pemutaran rentang surah selesai.");
                        Utils.toast(ctx,"Pemutaran selesai.",false);
                        if(listener!=null) listener.onUpdate(); return;
                    }
                    Utils.speak("Pembacaan Al-Qur'an selesai."); return;
                }
                na=1;
            }
        } else {
            if("ayat".equals(mode)){ final int rS=state.surah; final int rA=state.ayat; if(storage.getTtsEnabled()){ String t2=TerjemahanCache.get(rS,rA); if(t2!=null && t2.length()>0){ Utils.speakWithCallback(t2, new Utils.OnDone(){ public void run(){ play(rS,rA); } }); } else { play(rS,rA); } } else { play(rS,rA); } return; }
            else if("surah".equals(mode) && state.ayat>=Data.SURAH[state.surah-1].verses){ ns=state.surah; na=1; }
            else if("juz".equals(mode)){
                int curJuz=getJuzIndex(state.surah,state.ayat);
                int[] na2=getNextPosition(state.surah,state.ayat);
                int nextJuz=getJuzIndex(na2[0],na2[1]);
                if(curJuz!=nextJuz){ Data.JuzStart js=Data.JUZ_START[curJuz-1]; ns=js.surah; na=js.ayat; }
                else { ns=na2[0]; na=na2[1]; }
            } else { int[] np=getNextPosition(state.surah,state.ayat); ns=np[0]; na=np[1]; }
        }
        final int fNs=ns; final int fNa=na; if(storage.getTtsEnabled()){ String ter=TerjemahanCache.get(state.surah,state.ayat); if(ter!=null && ter.length()>0){ Utils.speakWithCallback(ter, new Utils.OnDone(){ public void run(){ play(fNs,fNa); } }); } else { play(fNs,fNa); } } else { play(fNs,fNa); }
    }
}

package com.kausar.audiokuran;

import android.os.Handler;
import android.os.Looper;
import org.json.JSONArray;
import org.json.JSONObject;
import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.net.HttpURLConnection;
import java.net.URL;
import java.util.HashMap;

public class TerjemahanCache {
    public interface Callback { void onReady(boolean ok); }

    private static HashMap<String,String> cache = new HashMap<>();
    private static HashMap<Integer,Boolean> loading = new HashMap<>();

    public static String get(int surah, int ayat){
        return cache.get(surah + ":" + ayat);
    }

    public static boolean isLoaded(int surah){
        return cache.containsKey(surah + ":1");
    }

    public static void fetch(final int surah, final Callback cb){
        if(isLoaded(surah)){ if(cb!=null) cb.onReady(true); return; }
        if(loading.containsKey(surah)){ if(cb!=null) cb.onReady(false); return; }
        loading.put(surah, true);
        new Thread(new Runnable(){
            public void run(){
                boolean ok = false;
                HttpURLConnection conn = null;
                try {
                    URL url = new URL("https://equran.id/api/v2/surat/" + surah);
                    conn = (HttpURLConnection) url.openConnection();
                    conn.setConnectTimeout(10000);
                    conn.setReadTimeout(15000);
                    conn.setRequestMethod("GET");
                    BufferedReader br = new BufferedReader(new InputStreamReader(conn.getInputStream(), "UTF-8"));
                    StringBuilder sb = new StringBuilder();
                    String line;
                    while((line = br.readLine()) != null) sb.append(line);
                    br.close();
                    JSONObject root = new JSONObject(sb.toString());
                    JSONObject data = root.getJSONObject("data");
                    JSONArray ayatArr = data.getJSONArray("ayat");
                    for(int i=0;i<ayatArr.length();i++){
                        JSONObject a = ayatArr.getJSONObject(i);
                        int no = a.getInt("nomorAyat");
                        String ter = a.getString("teksIndonesia");
                        cache.put(surah + ":" + no, ter);
                    }
                    ok = true;
                } catch(Exception e){
                    ok = false;
                } finally {
                    if(conn!=null) try { conn.disconnect(); } catch(Exception e){}
                    loading.remove(surah);
                }
                final boolean fOk = ok;
                new Handler(Looper.getMainLooper()).post(new Runnable(){
                    public void run(){ if(cb!=null) cb.onReady(fOk); }
                });
            }
        }).start();
    }
}

package com.kausar.audiokuran;

import android.content.Context;
import android.os.Bundle;
import android.speech.tts.TextToSpeech;
import android.speech.tts.UtteranceProgressListener;
import android.widget.Toast;
import java.util.Locale;

public class Utils {
    public static TextToSpeech tts;
    private static String lastUtteranceId = "quran";

    public interface OnDone { void run(); }
    private static OnDone pendingDone = null;

    public static void initTTS(Context ctx){
        if(tts==null){
            tts = new TextToSpeech(ctx, status -> {
                if(status==TextToSpeech.SUCCESS){
                    tts.setLanguage(Locale.getDefault());
                    tts.setOnUtteranceProgressListener(new UtteranceProgressListener(){
                        public void onStart(String id){}
                        public void onDone(String id){
                            if(pendingDone!=null){
                                OnDone d = pendingDone;
                                pendingDone = null;
                                new android.os.Handler(android.os.Looper.getMainLooper()).post(d::run);
                            }
                        }
                        public void onError(String id){
                            if(pendingDone!=null){
                                OnDone d = pendingDone;
                                pendingDone = null;
                                new android.os.Handler(android.os.Looper.getMainLooper()).post(d::run);
                            }
                        }
                    });
                }
            });
        }
    }

    public static void speak(String t){
        try { if(tts!=null && t!=null && !t.isEmpty()) tts.speak(t, TextToSpeech.QUEUE_FLUSH, null, "quran"); } catch(Exception e){}
    }

    public static void speakWithCallback(String t, OnDone done){
        try {
            if(tts==null || t==null || t.isEmpty()){ if(done!=null) done.run(); return; }
            pendingDone = done;
            tts.speak(t, TextToSpeech.QUEUE_FLUSH, null, "quran_cb");
        } catch(Exception e){
            pendingDone = null;
            if(done!=null) done.run();
        }
    }

    public static void toast(Context c, String t, boolean isLong){
        try { Toast.makeText(c, t, isLong?Toast.LENGTH_LONG:Toast.LENGTH_SHORT).show(); } catch(Exception e){}
    }
}
